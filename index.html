function formatRuntime(value){ if(value === undefined || value === null || value === ""){ return "---"; } let hours = Number(value); if(isNaN(hours) || hours <= 0){ return "---"; } let totalMinutes = Math.round(hours * 60); let h = Math.floor(totalMinutes / 60); let m = totalMinutes % 60; if(h > 0 && m > 0){ return h + "h " + m + "m"; } else if(h > 0){ return h + "h"; } else{ return m + "m"; } } function updateBattery(data){ let soc = Number(data.soc); let fill = document.getElementById("batteryFill"); let pct = document.getElementById("batteryPercent"); let stat = document.getElementById("batteryStatus"); // SOC DISPLAY if(!isNaN(soc)){ fill.style.width = soc + "%"; pct.innerText = soc.toFixed(0) + "%"; fill.className = "battery-fill"; if(soc < 20){ fill.classList.add("low"); } } let current = Number(data.current); let runtime = 0; let mode = "IDLE"; /* CHARGING Current positive */ if(current > 0.5){ mode = "CHARGING"; fill.classList.add( "charging" ); let remaining = Number(data.remaining_ah); let full = Number(data.full_ah); if( !isNaN(remaining) && !isNaN(full) ){ let needed = full - remaining; if(needed > 0){ runtime = needed / current; } } } /* DISCHARGING Current negative */ else if(current < -0.5){ mode = "DISCHARGING"; fill.classList.add( "discharging" ); let remaining = Number(data.remaining_ah); let load = Math.abs(current); if( !isNaN(remaining) && load > 0 ){ runtime = remaining / load; } } else{ mode = "IDLE"; } stat.innerText = mode; let result = formatRuntime(runtime); document.getElementById( "runtime" ).innerText = result; document.getElementById( "runtime2" ).innerText = result; // Change label let label = document.querySelector( ".battery-container div:nth-child(5)" ); if(label){ if(mode === "CHARGING"){ label.innerHTML = "Time To Full: "; } else if(mode === "DISCHARGING"){ label.innerHTML = "Runtime Left: "; } else{ label.innerHTML = "Remaining Time: "; } } }
